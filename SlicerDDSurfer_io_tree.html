SlicerDDSurfer 输入输出文件树（仅文件树）

输入树（统一输入契约） subject_root / input
  • <subject_root>/ # 单个受试者根目录
    • input/ # 统一输入契约目录
      • data.nii.gz # DWI 4D NIfTI
      • bvals # b-value
      • bvecs # gradient direction
      • nodif_brain_mask.nii.gz # brain mask
      • data.json # 可选：BIDS metadata
      • T1w.nii.gz # 可选：未来扩展，不是主流程前置
输出树（正式产物） subject_root / outputs
  • <subject_root>/outputs/ # 正式输出产物目录（稳定可复用）
    • 00_manifest/ # 全流程元数据与追溯索引
      • job_spec.json # 任务配置快照（参数节点导出）
      • pipeline_config.json # 各阶段参数与开关配置
      • pipeline_status.json # 各阶段运行状态（done/failed/skipped）
      • provenance.json # 输入输出、命令、版本、时间戳追溯
      • software_versions.json # Slicer 与外部引擎版本记录
      • output_file_index.tsv # 正式输出文件索引表
    • 01_preproc/ # 可选 DWI 预处理主产物目录；支持 DIPY 轻量后端或 MRtrix/FSL 完整后端，用户也可跳过
      • subj_dwi_final.nii.gz # 预处理后统一 DWI 主输入；跳过预处理时不存在
      • subj_dwi_final.bval # 与 final DWI 对应的 bval
      • subj_dwi_final.bvec # 与 final DWI 对应的 bvec
      • subj_dwi_final.nhdr # SlicerDMRI 友好的 DWI 头文件
      • subj_dwi_brain_mask.nii.gz # 与 final DWI 对齐的脑掩膜
      • subj_preproc_qc.json # 预处理质量控制摘要
    • 02_microstructure/ # 微结构模型结果目录
      • dti/ # DTI 标量结果；由 native DTI cache 真实生成
        • subj_dti_FA.nii.gz # FA 图
        • subj_dti_MD.nii.gz # MD 图
        • subj_dti_AD.nii.gz # AD 图；等同 L1 / 最大特征值
        • subj_dti_RD.nii.gz # RD 图；由 L2 和 L3 平均得到
        • subj_dti_L1.nii.gz # 第一特征值图
        • subj_dti_L2.nii.gz # 第二特征值图
        • subj_dti_L3.nii.gz # 第三特征值图
        • subj_dti_Trace.nii.gz # tensor trace 图
        • subj_dti_qc.json # DTI 拟合质量控制
      • freewater/ # Free-water 结果
        • subj_fw_fraction.nii.gz # Free-water fraction 图
        • subj_fw_FA.nii.gz # Free-water 校正 FA
        • subj_fw_MD.nii.gz # Free-water 校正 MD
        • subj_freewater_qc.json # Free-water 质量控制
      • noddi/ # NODDI 结果
        • subj_noddi_NDI.nii.gz # NODDI NDI 图
        • subj_noddi_ODI.nii.gz # NODDI ODI 图
        • subj_noddi_IsoVF.nii.gz # NODDI IsoVF 图
        • subj_noddi_qc.json # NODDI 拟合质量控制
    • 03_structural/ # 结构、表面、分区相关结果
      • ddseg/ # DDSeg 输出目录
        • subj_ddseg_labels.nii.gz # DDSeg 组织分割标签图
        • subj_ddseg_qc.json # DDSeg 质量控制
      • ddsurfer/ # DDSurfer 输出目录
        • surf/ # 表面网格结果
          • lh.white.vtk # 左半球 white surface
          • rh.white.vtk # 右半球 white surface
          • lh.pial.vtk # 左半球 pial surface
          • rh.pial.vtk # 右半球 pial surface
        • postprocess_annot/ # 后处理 annotation 结果
          • lh.aparc.annot # 左半球 aparc 注释
          • rh.aparc.annot # 右半球 aparc 注释
          • lh.DKT.annot # 左半球 DKT 注释
          • rh.DKT.annot # 右半球 DKT 注释
          • lh.Yeo7.annot # 左半球 Yeo7 注释
          • rh.Yeo7.annot # 右半球 Yeo7 注释
          • lh.Yeo17.annot # 左半球 Yeo17 注释
          • rh.Yeo17.annot # 右半球 Yeo17 注释
      • ddparcel/ # DDParcel 主产物目录（wmparc）
        • subj_ddparcel_wmparc_labels.nii.gz # DDParcel wmparc 标签图
        • subj_ddparcel_wmparc_lut.tsv # wmparc 标签查找表
        • subj_ddparcel_qc.json # DDParcel 质量控制
    • 04_tractography/ # 纤维追踪相关结果目录
      • fod/ # FOD 与 response function 结果
        • subj_response_wm.txt # WM response function
        • subj_response_gm.txt # GM response function
        • subj_response_csf.txt # CSF response function
        • subj_wmfod.nii.gz # WM FOD 图
        • subj_wmfod_norm.nii.gz # 强度归一化 WM FOD 图
      • act_prep/ # ACT 准备输入目录
        • subj_5tt.nii.gz # ACT 准备输入：5TT
        • subj_gmwmi_seed.nii.gz # ACT 准备输入：GMWMI 种子图
        • subj_act_qc.json # ACT 准备层质量控制
      • tractograms/ # 各方法全脑 tractogram 输出
        • dti_det/ # DTI deterministic tractogram 子目录
        • csd_prob/ # CSD probabilistic tractogram 子目录
        • csd_act/ # CSD + ACT tractogram 子目录
        • subj_tractograms_manifest.tsv # tractogram 方法、种子、权重索引
        • subj_tractography_qc.json # tractography 质量控制
      • saf/ # SAF short association fibres 提取结果；Python SAF 后端输出
        • SAF.tck # 短程联合纤维 tractogram 主输出
        • saf_manifest.json # SAF 参数、输入、依赖和输出追溯
        • saf_qc.json # SAF 后端可用性和输出完整性 QC
        • log.txt # SAF backend 运行日志
        • surf_data/ # SAF 投影到表面的曲率/统计文件
      • bundles/ # 束分割输出目录
        • subj_bundles_manifest.tsv # 束分割来源与方法索引
        • subj_bundles_qc.json # 束分割质量控制
    • 05_connectome/ # 连接组矩阵与边表目录
      • subj_desc-csdAct_wmparc_connectome_counts.csv # 基于 CSD+ACT 的 counts 矩阵
      • subj_desc-csdAct_wmparc_connectome_sift2.csv # 基于 CSD+ACT 的 SIFT2 加权矩阵
      • subj_connectome_nodes_wmparc.tsv # wmparc 节点信息表
      • subj_connectome_edges_wmparc.tsv # 边列表（长表）
      • subj_connectome_manifest.tsv # connectome 版本、来源 tractogram 索引
      • subj_connectome_qc.json # connectome 质量控制
    • 06_statistics/ # 区域级与束级统计结果
      • regional_microstructure/ # 脑区级微结构统计结果目录
      • bundle_profiles/ # 束级 tractometry profile 目录
      • subj_group_ready_features.tsv # 群体分析可直接拼接的宽表
      • subj_feature_dictionary.tsv # 特征字段字典与单位说明
    • 07_slicer_scene_report/ # 场景归档与可视化报告目录
      • subj_slicer_scene.mrb # Slicer 场景归档
      • subj_qc_report.html # 可浏览质量控制报告
      • figures/ # 报告引用截图与图表目录
缓存树（可重建临时层） subject_root / cache
  • <subject_root>/cache/ # 可重建临时缓存层
    • mrtrix_work/ # MRtrix 相关临时工作目录
      • dwifslpreproc_tmp/ # dwifslpreproc 临时目录
      • response_tmp/ # response function 估计临时目录
      • tckgen_tmp/ # tractography 生成临时目录
    • optional_alignment_work/ # 可选对齐扩展临时目录
      • alignment_tmp/ # 可选对齐扩展临时目录
    • deep_models/ # 深度模型中间输出缓存
      • ddseg_logits.nii.gz # DDSeg 中间 logits
      • ddsurfer_raw_surfaces/ # DDSurfer 原始 mesh 中间目录
      • ddparcel_raw_probs.nii.gz # DDParcel 中间概率图
    • cache_manifest.json # 缓存层索引与清理依据
